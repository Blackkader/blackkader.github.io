def oid_to_hex(oid):

    components = list(map(int, oid.split('.')))

    hex_oid = [(components[0] * 40) + components[1]]

    for component in components[2:]:
        hex_part = []
        while component > 0:
            hex_part.insert(0, component & 0x7f)
            component >>= 7
        hex_oid.extend(hex_part)

    return ''.join(f'{byte:02x}' for byte in hex_oid)
